const audio = document.getElementById('audioPlayer'); const playBtn = document.getElementById('playBtn'); const currentTimeEl = document.getElementById('currentTime'); const remainingTimeEl = document.getElementById('remainingTime'); const volumeSlider = document.getElementById('volumeSlider'); // Modal Elements const instructionModal = document.getElementById('instructionModal'); const closeModalBtn = document.getElementById('closeModalBtn'); // Close Modal Logic closeModalBtn.addEventListener('click', () => { instructionModal.classList.add('hidden'); }); // Time Formatter function formatTime(seconds) { if (isNaN(seconds) || !isFinite(seconds) || seconds < 0) return "0:00"; const hrs = Math.floor(seconds / 3600); const min = Math.floor((seconds % 3600) / 60); const sec = Math.floor(seconds % 60); const formattedMin = hrs > 0 && min < 10 ? '0' + min : min; const formattedSec = sec < 10 ? '0' + sec : sec; return hrs > 0 ? `${hrs}:${formattedMin}:${formattedSec}` : `${formattedMin}:${formattedSec}`; } // Load LocalStorage Data window.addEventListener('load', () => { const savedTime = localStorage.getItem('audio2_progress'); if (savedTime !== null) { audio.currentTime = parseFloat(savedTime); } const savedVolume = localStorage.getItem('audio2_volume'); if (savedVolume !== null) { audio.volume = parseFloat(savedVolume); volumeSlider.value = savedVolume; } }); audio.addEventListener('loadedmetadata', () => { currentTimeEl.textContent = formatTime(audio.currentTime); remainingTimeEl.textContent = "-" + formatTime(audio.duration - audio.currentTime); }); // Play Only Logic with Error Handling playBtn.addEventListener('click', () => { if (audio.paused) { // Safety check: if progress is stuck at the end, restart it if (audio.currentTime >= audio.duration) { audio.currentTime = 0; } const playPromise = audio.play(); if (playPromise !== undefined) { playPromise.then(() => { // Playback started successfully playBtn.textContent = 'Playing...'; playBtn.disabled = true; }).catch(error => { // Playback failed console.error("Playback failed:", error); alert("Cannot play audio. Please verify 'audio2.mp3' is in the exact same folder as this file."); playBtn.textContent = 'Play'; playBtn.disabled = false; }); } } }); // Time Update & Save Progress audio.addEventListener('timeupdate', () => { currentTimeEl.textContent = formatTime(audio.currentTime); if (!isNaN(audio.duration) && isFinite(audio.duration)) { remainingTimeEl.textContent = "-" + formatTime(audio.duration - audio.currentTime); } localStorage.setItem('audio2_progress', audio.currentTime); }); // Volume Update volumeSlider.addEventListener('input', (e) => { audio.volume = e.target.value; localStorage.setItem('audio2_volume', audio.volume); }); // Reset when finished audio.addEventListener('ended', () => { playBtn.textContent = 'Play'; playBtn.disabled = false; localStorage.removeItem('audio2_progress'); audio.currentTime = 0; currentTimeEl.textContent = "0:00"; remainingTimeEl.textContent = "-" + formatTime(audio.duration); });
